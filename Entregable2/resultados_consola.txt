> # Función para calcular AP para una clase vs resto
> calculate_ap <- function(y_true, y_pred_proba, target_class) {
+   # Validaciones iniciales
+   if(is.null(y_pred_proba) || !is.matrix(y_pred_proba)) {
+     return(NA)
+   }
+   
+   n_cols <- ncol(y_pred_proba)
+   if(is.null(n_cols) || n_cols == 0) {
+     return(NA)
+   }
+   
+   # Convertir a binario: 1 si es target_class, 0 si no
+   y_binary <- as.numeric(y_true == target_class)
+   
+   # Obtener probabilidades para esta clase
+   class_idx <- target_class + 1  # R indexa desde 1
+   if(class_idx > n_cols) {
+     return(NA)
+   }
+   y_score <- y_pred_proba[, class_idx]
+   
+   # Calcular AP usando PRROC
+   if(sum(y_binary == 1) == 0 || sum(y_binary == 0) == 0) {
+     return(NA)  # No hay ejemplos de esta clase
+   }
+   
+   pr_result <- tryCatch({
+     pr.curve(scores.class0 = y_score[y_binary == 1],
+              scores.class1 = y_score[y_binary == 0],
+              curve = FALSE)
+   }, error = function(e) {
+     return(list(auc.integral = NA))
+   })
+   
+   return(pr_result$auc.integral)
+ }
